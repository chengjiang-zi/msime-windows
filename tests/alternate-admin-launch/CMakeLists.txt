cmake_minimum_required(VERSION 3.25)
project(PersonalAlternateAdminProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This probe requires a disposable Windows runner")
endif()
add_executable(personal-alternate-admin-launch main.cpp)
target_compile_features(personal-alternate-admin-launch PRIVATE cxx_std_17)
target_compile_definitions(personal-alternate-admin-launch PRIVATE NOMINMAX UNICODE _UNICODE)
target_link_libraries(personal-alternate-admin-launch PRIVATE advapi32 userenv)
if(MSVC)
    target_compile_options(personal-alternate-admin-launch PRIVATE /utf-8 /W4)
endif()
