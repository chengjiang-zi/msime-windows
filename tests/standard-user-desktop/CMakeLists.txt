cmake_minimum_required(VERSION 3.25)
project(PersonalStandardDesktopProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This probe requires a disposable Windows runner")
endif()
add_executable(personal-standard-desktop-launch main.cpp)
add_executable(personal-standard-input-probe input_probe.cpp)
target_compile_features(personal-standard-desktop-launch PRIVATE cxx_std_17)
target_compile_features(personal-standard-input-probe PRIVATE cxx_std_17)
target_compile_definitions(personal-standard-desktop-launch PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_definitions(personal-standard-input-probe PRIVATE NOMINMAX UNICODE _UNICODE)
target_link_libraries(personal-standard-desktop-launch PRIVATE advapi32 userenv)
target_link_libraries(personal-standard-input-probe PRIVATE crypt32)
if(MSVC)
    target_compile_options(personal-standard-desktop-launch PRIVATE /utf-8 /W4)
    target_compile_options(personal-standard-input-probe PRIVATE /utf-8 /W4)
endif()
