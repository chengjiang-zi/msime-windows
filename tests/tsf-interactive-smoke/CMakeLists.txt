cmake_minimum_required(VERSION 3.25)
project(PersonalTsfInteractiveSmoke LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This smoke test requires a disposable Windows desktop")
endif()
add_executable(personal-tsf-interactive-smoke main.cpp)
target_compile_features(personal-tsf-interactive-smoke PRIVATE cxx_std_17)
target_compile_definitions(personal-tsf-interactive-smoke PRIVATE NOMINMAX UNICODE _UNICODE)
target_link_libraries(personal-tsf-interactive-smoke PRIVATE ole32 uuid user32)
if(MSVC)
    target_compile_options(personal-tsf-interactive-smoke PRIVATE /utf-8 /W4)
endif()
