cmake_minimum_required(VERSION 3.25)
project(PersonalLimitedTokenProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This probe requires a disposable Windows runner")
endif()
add_executable(personal-limited-token-launch main.cpp)
target_compile_features(personal-limited-token-launch PRIVATE cxx_std_17)
target_compile_definitions(personal-limited-token-launch PRIVATE NOMINMAX UNICODE _UNICODE)
target_link_libraries(personal-limited-token-launch PRIVATE advapi32)
if(MSVC)
    target_compile_options(personal-limited-token-launch PRIVATE /utf-8 /W4)
endif()
